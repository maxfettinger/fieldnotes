{{- /* The look in effect on a page: the preset (how things are drawn) and the colour scheme (its colours, in a light and a dark version). From the page's front matter (style: { preset, scheme }), else the site's params.style, else the preset's own scheme (data/fieldnotes/presets/.toml). Unknown names fall back to fieldnotes, with a warning. Returns a dict: preset, scheme, fonts (the preset's Google Fonts specs). Call it with partialCached and the page's style as the variant key (see head.html). */ -}} {{- $site := site.Params.style | default dict }} {{- $page := .Params.style | default dict }} {{- $presets := hugo.Data.fieldnotes.presets }} {{- $name := $page.preset | default $site.preset | default "fieldnotes" }} {{- $preset := index $presets $name }} {{- if not $preset }} {{- warnf "fieldnotes: preset %q not found in data/fieldnotes/presets/, using fieldnotes" $name }} {{- $name = "fieldnotes" }} {{- $preset = index $presets "fieldnotes" }} {{- end }} {{- $scheme := $page.scheme | default $site.scheme | default $preset.scheme | default "fieldnotes" }} {{- if not (index hugo.Data.fieldnotes.schemes $scheme) }} {{- warnf "fieldnotes: scheme %q not found in data/fieldnotes/schemes/, using fieldnotes" $scheme }} {{- $scheme = "fieldnotes" }} {{- end }} {{- return dict "preset" $name "scheme" $scheme "fonts" ($preset.fonts | default slice) }}
